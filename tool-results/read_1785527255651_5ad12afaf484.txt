     1→"use client";
     2→
     3→import { useState, useCallback, useEffect } from "react";
     4→import Link from "next/link";
     5→
     6→type Step = 1 | 2 | 3 | 4 | 5;
     7→
     8→const stepLabels: Record<Step, string> = {
     9→  1: "Case details",
    10→  2: "Consent",
    11→  3: "Statement upload",
    12→  4: "Processing",
    13→  5: "Review",
    14→};
    15→
    16→interface CaseDetails {
    17→  lenderReference: string;
    18→  applicantDisplayName: string;
    19→  productType: string;
    20→  requestedAmount: string;
    21→  assessmentPurpose: string;
    22→}
    23→
    24→interface ConsentDetails {
    25→  purposeAccepted: boolean;
    26→  dataCategoriesAccepted: boolean;
    27→  organisationAccepted: boolean;
    28→  retentionAccepted: boolean;
    29→  sharingScopeAccepted: boolean;
    30→  expiryAccepted: boolean;
    31→}
    32→
    33→const productTypes = [
    34→  "Personal loan",
    35→  "Microfinance loan",
    36→  "Retail credit",
    37→  "Vehicle finance",
    38→  "Business loan",
    39→  "Housing loan",
    40→];
    41→
    42→const assessmentPurposes = [
    43→  "New credit application",
    44→  "Credit limit increase",
    45→  "Account review",
    46→  "Debt restructuring",
    47→  "Regulatory compliance",
    48→];
    49→
    50→export default function NewApplicationPage() {
    51→  const [currentStep, setCurrentStep] = useState<Step>(1);
    52→  const [caseDetails, setCaseDetails] = useState<CaseDetails>({
    53→    lenderReference: "",
    54→    applicantDisplayName: "",
    55→    productType: "",
    56→    requestedAmount: "",
    57→    assessmentPurpose: "",
    58→  });
    59→  const [consent, setConsent] = useState<ConsentDetails>({
    60→    purposeAccepted: false,
    61→    dataCategoriesAccepted: false,
    62→    organisationAccepted: false,
    63→    retentionAccepted: false,
    64→    sharingScopeAccepted: false,
    65→    expiryAccepted: false,
    66→  });
    67→  const [uploadedFiles, setUploadedFiles] = useState<{ name: string; size: string; type: string }[]>([]);
    68→  const [isDragging, setIsDragging] = useState(false);
    69→  const [uploadProgress, setUploadProgress] = useState(0);
    70→  const [isUploading, setIsUploading] = useState(false);
    71→  const [processingStage, setProcessingStage] = useState(0);
    72→  const [processingComplete, setProcessingComplete] = useState(false);
    73→
    74→  const allConsentAccepted = Object.values(consent).every(Boolean);
    75→  const caseDetailsValid =
    76→    caseDetails.lenderReference.trim() !== "" &&
    77→    caseDetails.applicantDisplayName.trim() !== "" &&
    78→    caseDetails.productType !== "" &&
    79→    caseDetails.requestedAmount.trim() !== "" &&
    80→    caseDetails.assessmentPurpose !== "";
    81→
    82→  // Simulate upload progress
    83→  useEffect(() => {
    84→    if (currentStep === 3 && isUploading) {
    85→      const interval = setInterval(() => {
    86→        setUploadProgress((prev) => {
    87→          if (prev >= 100) {
    88→            setIsUploading(false);
    89→            clearInterval(interval);
    90→            return 100;
    91→          }
    92→          return prev + 5;
    93→        });
    94→      }, 100);
    95→      return () => clearInterval(interval);
    96→    }
    97→  }, [currentStep, isUploading]);
    98→
    99→  // Simulate processing stages
   100→  useEffect(() => {
   101→    if (currentStep === 4 && !processingComplete) {
   102→      const stages = [
   103→        { delay: 800, stage: 1 },
   104→        { delay: 1600, stage: 2 },
   105→        { delay: 2400, stage: 3 },
   106→        { delay: 3200, stage: 4 },
   107→        { delay: 4000, stage: 5 },
   108→      ];
   109→      const timers = stages.map(({ delay, stage }) =>
   110→        setTimeout(() => setProcessingStage(stage), delay)
   111→      );
   112→      const completeTimer = setTimeout(() => setProcessingComplete(true), 4800);
   113→      return () => {
   114→        timers.forEach(clearTimeout);
   115→        clearTimeout(completeTimer);
   116→      };
   117→    }
   118→  }, [currentStep, processingComplete]);
   119→
   120→  const handleDragOver = useCallback((e: React.DragEvent) => {
   121→    e.preventDefault();
   122→    setIsDragging(true);
   123→  }, []);
   124→
   125→  const handleDragLeave = useCallback((e: React.DragEvent) => {
   126→    e.preventDefault();
   127→    setIsDragging(false);
   128→  }, []);
   129→
   130→  const handleDrop = useCallback((e: React.DragEvent) => {
   131→    e.preventDefault();
   132→    setIsDragging(false);
   133→    const files = Array.from(e.dataTransfer.files);
   134→    const validTypes = ["application/pdf", "image/png", "image/jpeg"];
   135→    const validFiles = files.filter((f) => validTypes.includes(f.type));
   136→    const newFiles = validFiles.map((f) => ({
   137→      name: f.name,
   138→      size: f.size > 1024 * 1024 ? `${(f.size / (1024 * 1024)).toFixed(1)} MB` : `${(f.size / 1024).toFixed(0)} KB`,
   139→      type: f.type.split("/")[1].toUpperCase(),
   140→    }));
   141→    setUploadedFiles((prev) => [...prev, ...newFiles]);
   142→    if (newFiles.length > 0) {
   143→      setIsUploading(true);
   144→      setUploadProgress(0);
   145→    }
   146→  }, []);
   147→
   148→  const handleFileInput = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
   149→    const files = Array.from(e.target.files || []);
   150→    const newFiles = files.map((f) => ({
   151→      name: f.name,
   152→      size: f.size > 1024 * 1024 ? `${(f.size / (1024 * 1024)).toFixed(1)} MB` : `${(f.size / 1024).toFixed(0)} KB`,
   153→      type: f.type.split("/")[1].toUpperCase(),
   154→    }));
   155→    setUploadedFiles((prev) => [...prev, ...newFiles]);
   156→    if (newFiles.length > 0) {
   157→      setIsUploading(true);
   158→      setUploadProgress(0);
   159→    }
   160→  }, []);
   161→
   162→  const removeFile = useCallback((index: number) => {
   163→    setUploadedFiles((prev) => prev.filter((_, i) => i !== index));
   164→  }, []);
   165→
   166→  const processingStages = [
   167→    { label: "Received", status: processingStage >= 1 ? "complete" as const : "pending" as const },
   168→    { label: "Security checks", status: processingStage >= 2 ? "complete" as const : processingStage === 1 ? "active" as const : "pending" as const },
   169→    { label: "Text extraction", status: processingStage >= 3 ? "complete" as const : processingStage === 2 ? "active" as const : "pending" as const },
   170→    { label: "Transaction review", status: processingStage >= 4 ? "complete" as const : processingStage === 3 ? "active" as const : "pending" as const },
   171→    { label: "Assessment calculation", status: processingStage >= 5 ? "complete" as const : processingStage === 4 ? "active" as const : "pending" as const },
   172→    { label: "Ready", status: processingComplete ? "complete" as const : processingStage === 5 ? "active" as const : "pending" as const },
   173→  ];
   174→
   175→  return (
   176→    <div className="max-w-3xl mx-auto space-y-8">
   177→      {/* Header */}
   178→      <div>
   179→        <div className="flex items-center gap-2 text-sm text-ink/50 mb-2">
   180→          <Link href="/app/applications" className="hover:text-ink transition-colors duration-ui">Applications</Link>
   181→          <svg width="12" height="12" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M9 18l6-6-6-6" /></svg>
   182→          <span className="text-ink">New application</span>
   183→        </div>
   184→        <div className="flex items-center gap-2">
   185→          <h1 className="text-2xl font-bold text-ink">New credit assessment</h1>
   186→          <span className="inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium bg-teal-50 text-teal-500 border border-teal-200">
   187→            Demo
   188→          </span>
   189→        </div>
   190→        <p className="mt-1 text-sm text-ink/60">
   191→          Complete each step to submit a new cash-flow assessment application. All data is synthetic.
   192→        </p>
   193→      </div>
   194→
   195→      {/* Step progress indicator */}
   196→      <nav aria-label="Application progress" className="relative">
   197→        <div className="flex items-center justify-between">
   198→          {([1, 2, 3, 4, 5] as Step[]).map((step) => (
   199→            <div key={step} className="flex flex-col items-center relative z-10">
   200→              <div
   201→                className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-medium transition-colors duration-ui ${
   202→                  step < currentStep
   203→                    ? "bg-teal-400 text-white"
   204→                    : step === currentStep
   205→                    ? "bg-ink text-sand-50"
   206→                    : "bg-sand-300 text-ink/40"
   207→                }`}
   208→              >
   209→                {step < currentStep ? (
   210→                  <svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round">
   211→                    <path d="M20 6L9 17l-5-5" />
   212→                  </svg>
   213→                ) : (
   214→                  step
   215→                )}
   216→              </div>
   217→              <span className={`mt-1.5 text-xs font-medium hidden sm:block ${
   218→                step <= currentStep ? "text-ink" : "text-ink/40"
   219→              }`}>
   220→                {stepLabels[step]}
   221→              </span>
   222→            </div>
   223→          ))}
   224→        </div>
   225→        <div className="absolute top-4 left-0 right-0 h-0.5 bg-sand-300 -z-0" aria-hidden="true">
   226→          <div
   227→            className="h-full bg-ink transition-all duration-500"
   228→            style={{ width: `${((currentStep - 1) / 4) * 100}%` }}
   229→          />
   230→        </div>
   231→      </nav>
   232→
   233→      {/* Step content */}
   234→      <div className="bg-white border border-sand-300 rounded-lg p-6 sm:p-8">
   235→        {/* Step 1: Case details */}
   236→        {currentStep === 1 && (
   237→          <div className="space-y-6">
   238→            <div>
   239→              <h2 className="text-lg font-semibold text-ink">Case details</h2>
   240→              <p className="mt-1 text-sm text-ink/60">Provide the basic details for this credit assessment.</p>
   241→            </div>
   242→            <div className="space-y-4">
   243→              <div>
   244→                <label htmlFor="lenderReference" className="block text-sm font-medium text-ink mb-1.5">
   245→                  Lender reference <span className="text-alert">*</span>
   246→                </label>
   247→                <input
   248→                  id="lenderReference"
   249→                  type="text"
   250→                  value={caseDetails.lenderReference}
   251→                  onChange={(e) => setCaseDetails((prev) => ({ ...prev, lenderReference: e.target.value }))}
   252→                  placeholder="e.g. LN-2024-00142"
   253→                  className="w-full px-3 py-2 border border-sand-300 rounded-md text-sm text-ink placeholder:text-ink/30 focus:outline-none focus:ring-2 focus:ring-teal-400 focus:border-transparent transition-colors duration-ui"
   254→                />
   255→              </div>
   256→              <div>
   257→                <label htmlFor="applicantName" className="block text-sm font-medium text-ink mb-1.5">
   258→                  Applicant display name <span className="text-alert">*</span>
   259→                </label>
   260→                <input
   261→                  id="applicantName"
   262→                  type="text"
   263→                  value={caseDetails.applicantDisplayName}
   264→                  onChange={(e) => setCaseDetails((prev) => ({ ...prev, applicantDisplayName: e.target.value }))}
   265→                  placeholder="e.g. Maria K."
   266→                  className="w-full px-3 py-2 border border-sand-300 rounded-md text-sm text-ink placeholder:text-ink/30 focus:outline-none focus:ring-2 focus:ring-teal-400 focus:border-transparent transition-colors duration-ui"
   267→                />
   268→                <p className="mt-1 text-xs text-ink/50">A display name is used instead of the full legal name to protect borrower privacy.</p>
   269→              </div>
   270→              <div>
   271→                <label htmlFor="productType" className="block text-sm font-medium text-ink mb-1.5">
   272→                  Product type <span className="text-alert">*</span>
   273→                </label>
   274→                <select
   275→                  id="productType"
   276→                  value={caseDetails.productType}
   277→                  onChange={(e) => setCaseDetails((prev) => ({ ...prev, productType: e.target.value }))}
   278→                  className="w-full px-3 py-2 border border-sand-300 rounded-md text-sm text-ink focus:outline-none focus:ring-2 focus:ring-teal-400 focus:border-transparent transition-colors duration-ui appearance-none bg-white"
   279→                >
   280→                  <option value="">Select a product type</option>
   281→                  {productTypes.map((type) => (
   282→                    <option key={type} value={type}>{type}</option>
   283→                  ))}
   284→                </select>
   285→              </div>
   286→              <div>
   287→                <label htmlFor="requestedAmount" className="block text-sm font-medium text-ink mb-1.5">
   288→                  Requested amount (NAD) <span className="text-alert">*</span>
   289→                </label>
   290→                <input
   291→                  id="requestedAmount"
   292→                  type="text"
   293→                  value={caseDetails.requestedAmount}
   294→                  onChange={(e) => setCaseDetails((prev) => ({ ...prev, requestedAmount: e.target.value }))}
   295→                  placeholder="e.g. 15000"
   296→                  className="w-full px-3 py-2 border border-sand-300 rounded-md text-sm text-ink placeholder:text-ink/30 focus:outline-none focus:ring-2 focus:ring-teal-400 focus:border-transparent transition-colors duration-ui"
   297→                />
   298→              </div>
   299→              <div>
   300→                <label htmlFor="assessmentPurpose" className="block text-sm font-medium text-ink mb-1.5">
   301→                  Assessment purpose <span className="text-alert">*</span>
   302→                </label>
   303→                <select
   304→                  id="assessmentPurpose"
   305→                  value={caseDetails.assessmentPurpose}
   306→                  onChange={(e) => setCaseDetails((prev) => ({ ...prev, assessmentPurpose: e.target.value }))}
   307→                  className="w-full px-3 py-2 border border-sand-300 rounded-md text-sm text-ink focus:outline-none focus:ring-2 focus:ring-teal-400 focus:border-transparent transition-colors duration-ui appearance-none bg-white"
   308→                >
   309→                  <option value="">Select an assessment purpose</option>
   310→                  {assessmentPurposes.map((purpose) => (
   311→                    <option key={purpose} value={purpose}>{purpose}</option>
   312→                  ))}
   313→                </select>
   314→              </div>
   315→            </div>
   316→            <div className="flex justify-end">
   317→              <button
   318→                type="button"
   319→                disabled={!caseDetailsValid}
   320→                onClick={() => setCurrentStep(2)}
   321→                className="px-4 py-2 bg-ink text-sand-50 rounded-md text-sm font-medium hover:bg-ink-50 transition-colors duration-ui disabled:opacity-40 disabled:cursor-not-allowed"
   322→              >
   323→                Continue
   324→              </button>
   325→            </div>
   326→          </div>
   327→        )}
   328→
   329→        {/* Step 2: Consent */}
   330→        {currentStep === 2 && (
   331→          <div className="space-y-6">
   332→            <div>
   333→              <h2 className="text-lg font-semibold text-ink">Data processing consent</h2>
   334→              <p className="mt-1 text-sm text-ink/60">
   335→                The applicant must consent to the following before their data can be processed. Review each item and confirm.
   336→              </p>
   337→            </div>
   338→            <div className="space-y-4">
   339→              {[
   340→                { key: "purposeAccepted" as const, id: "consentPurpose", label: "Purpose of processing", desc: "The applicant's bank statement data will be processed solely for the purpose of conducting a cash-flow credit assessment for the product and purpose specified above." },
   341→                { key: "dataCategoriesAccepted" as const, id: "consentData", label: "Data categories", desc: "The following data categories will be processed: transaction amounts, dates, descriptions, and direction (credit/debit). No account numbers or personal identifiers will be stored." },
   342→                { key: "organisationAccepted" as const, id: "consentOrg", label: "Processing organisation", desc: "Data will be processed by Hola Credit on behalf of your organisation. Hola Credit acts as a data processor under your instructions." },
   343→                { key: "retentionAccepted" as const, id: "consentRetention", label: "Retention period", desc: "Raw statement data will be retained for 90 days after assessment completion, after which it will be permanently deleted. Assessment results are retained for 3 years." },
   344→                { key: "sharingScopeAccepted" as const, id: "consentSharing", label: "Sharing scope", desc: "Assessment results will be shared only with authorised members of your organisation. Hola Credit will not share results with third parties without your consent." },
   345→                { key: "expiryAccepted" as const, id: "consentExpiry", label: "Consent expiry", desc: "This consent expires 90 days from capture. If the assessment is not completed within this period, the data will be deleted and a new consent will be required." },
   346→              ].map((item) => (
   347→                <div key={item.key} className="border border-sand-300 rounded-md p-4">
   348→                  <div className="flex items-start gap-3">
   349→                    <input
   350→                      id={item.id}
   351→                      type="checkbox"
   352→                      checked={consent[item.key]}
   353→                      onChange={(e) => setConsent((prev) => ({ ...prev, [item.key]: e.target.checked }))}
   354→                      className="mt-0.5 w-4 h-4 rounded border-sand-300 text-teal-400 focus:ring-teal-400"
   355→                    />
   356→                    <div>
   357→                      <label htmlFor={item.id} className="text-sm font-medium text-ink cursor-pointer">{item.label}</label>
   358→                      <p className="mt-0.5 text-sm text-ink/60">{item.desc}</p>
   359→                    </div>
   360→                  </div>
   361→                </div>
   362→              ))}
   363→            </div>
   364→            <div className="flex items-center justify-between">
   365→              <button
   366→                type="button"
   367→                onClick={() => setCurrentStep(1)}
   368→                className="px-4 py-2 text-sm font-medium text-ink/60 hover:text-ink transition-colors duration-ui"
   369→              >
   370→                Back
   371→              </button>
   372→              <button
   373→                type="button"
   374→                disabled={!allConsentAccepted}
   375→                onClick={() => setCurrentStep(3)}
   376→                className="px-4 py-2 bg-ink text-sand-50 rounded-md text-sm font-medium hover:bg-ink-50 transition-colors duration-ui disabled:opacity-40 disabled:cursor-not-allowed"
   377→              >
   378→                Continue
   379→              </button>
   380→            </div>
   381→          </div>
   382→        )}
   383→
   384→        {/* Step 3: Statement upload */}
   385→        {currentStep === 3 && (
   386→          <div className="space-y-6">
   387→            <div>
   388→              <h2 className="text-lg font-semibold text-ink">Statement upload</h2>
   389→              <p className="mt-1 text-sm text-ink/60">
   390→                Upload the applicant&apos;s bank statement(s). Accepted formats: PDF, PNG, JPEG. Maximum file size: 20 MB.
   391→              </p>
   392→            </div>
   393→
   394→            {/* Drop zone */}
   395→            <div
   396→              onDragOver={handleDragOver}
   397→              onDragLeave={handleDragLeave}
   398→              onDrop={handleDrop}
   399→              className={`border-2 border-dashed rounded-lg p-8 text-center transition-colors duration-ui ${
   400→                isDragging ? "border-teal-400 bg-teal-50/50" : "border-sand-300 hover:border-sand-400"
   401→              }`}
   402→            >
   403→              <svg width="32" height="32" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" className="mx-auto text-ink/30 mb-3">
   404→                <path d="M21 15v4a2 2 0 01-2 2H5a2 2 0 01-2-2v-4M17 8l-5-5-5 5M12 3v12" />
   405→              </svg>
   406→              <p className="text-sm font-medium text-ink">Drag and drop files here</p>
   407→              <p className="mt-1 text-xs text-ink/50">or click to browse</p>
   408→              <input
   409→                type="file"
   410→                accept=".pdf,.png,.jpeg,.jpg"
   411→                multiple
   412→                onChange={handleFileInput}
   413→                className="absolute inset-0 w-full h-full opacity-0 cursor-pointer"
   414→                style={{ position: "relative", marginTop: "8px" }}
   415→              />
   416→            </div>
   417→
   418→            {/* Upload progress */}
   419→            {isUploading && (
   420→              <div className="space-y-2">
   421→                <div className="flex items-center justify-between text-sm">
   422→                  <span className="text-ink/60">Uploading...</span>
   423→                  <span className="font-medium text-ink">{uploadProgress}%</span>
   424→                </div>
   425→                <div className="w-full h-2 bg-sand-200 rounded-full overflow-hidden">
   426→                  <div className="h-full bg-teal-400 rounded-full transition-all duration-200" style={{ width: `${uploadProgress}%` }} />
   427→                </div>
   428→              </div>
   429→            )}
   430→
   431→            {/* Uploaded files */}
   432→            {uploadedFiles.length > 0 && (
   433→              <div className="space-y-2">
   434→                <h3 className="text-sm font-medium text-ink">Uploaded files</h3>
   435→                {uploadedFiles.map((file, index) => (
   436→                  <div key={index} className="flex items-center justify-between p-3 bg-sand-50 border border-sand-300 rounded-md">
   437→                    <div className="flex items-center gap-3">
   438→                      <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.75" strokeLinecap="round" strokeLinejoin="round" className="text-teal-500">
   439→                        <path d="M14 2H6a2 2 0 00-2 2v16a2 2 0 002 2h12a2 2 0 002-2V8z" /><path d="M14 2v6h6M16 13H8M16 17H8M10 9H8" />
   440→                      </svg>
   441→                      <div>
   442→                        <p className="text-sm font-medium text-ink">{file.name}</p>
   443→                        <p className="text-xs text-ink/50">{file.size} &middot; {file.type}</p>
   444→                      </div>
   445→                    </div>
   446→                    <button
   447→                      type="button"
   448→                      onClick={() => removeFile(index)}
   449→                      className="p-1 text-ink/40 hover:text-alert transition-colors duration-ui"
   450→                    >
   451→                      <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
   452→                        <path d="M6 18L18 6M6 6l12 12" />
   453→                      </svg>
   454→                    </button>
   455→                  </div>
   456→                ))}
   457→              </div>
   458→            )}
   459→
   460→            <div className="flex items-center justify-between">
   461→              <button
   462→                type="button"
   463→                onClick={() => setCurrentStep(2)}
   464→                className="px-4 py-2 text-sm font-medium text-ink/60 hover:text-ink transition-colors duration-ui"
   465→              >
   466→                Back
   467→              </button>
   468→              <button
   469→                type="button"
   470→                disabled={uploadedFiles.length === 0 || isUploading}
   471→                onClick={() => { setCurrentStep(4); setProcessingStage(0); setProcessingComplete(false); }}
   472→                className="px-4 py-2 bg-ink text-sand-50 rounded-md text-sm font-medium hover:bg-ink-50 transition-colors duration-ui disabled:opacity-40 disabled:cursor-not-allowed"
   473→              >
   474→                Submit for processing
   475→              </button>
   476→            </div>
   477→          </div>
   478→        )}
   479→
   480→        {/* Step 4: Processing */}
